          return 'নতুন ফ্যাক্ট: সঠিকভাবে বায়ুরোধী পাত্রে রাখলে মধু কখনো নষ্ট হয় না। এছাড়া অক্টোপাসের তিনটি হৃদপিণ্ড এবং নীল রঙের রক্ত থাকে।';
        }

        // Math response
        if (q.includes('গণিত') || q.includes('দ্বিঘাত') || q.includes('math') || q.includes('equation') || q.includes('x²')) {

        }

        // Science / Physics / Newton
        if (q.includes('বিজ্ঞান') || q.includes('নিউটন') || q.includes('science') || q.includes('newton') || q.includes('circuit') || q.includes('motor') || q.includes('led')) {

For an LED circuit, always connect a current-limiting resistor in series.';

        }

        if (replyLang === 'en') {

      function handleUserQuery(rawText, fromVoice = false) {
        const trimmed = String(rawText || '').trim();
        if (!trimmed) return;

        appendUserMessage(trimmed, fromVoice);
        dom.promptInput.value = '';
        dom.liveSpeechPreview.textContent = (fromVoice ? '🎙️ ' : '') + trimmed;

        // 1. Strict safety refusal rule
        if (isHarmfulRequest(trimmed)) {
          const replyLang = determineReplyLanguage(trimmed);
          appendAssistantMessage(REFUSAL_PHRASE, 'SAFETY RULE', replyLang);
          speakReply(REFUSAL_PHRASE, 'bn');
          return;
        }

        // 2. Bilingual mode switch voice/text command ("আমি ইংরেজি বলবো তুমি বাংলা বলবে")
        const normalized = trimmed.toLowerCase();
        if (
          normalized.includes('আমি ইংরেজি বলবো তুমি বাংলা বলবে') ||
          normalized.includes('তুমি বাংলা বলবে আমি ইংরেজি বলবো') ||
          normalized.includes('i speak english you reply in bangla') ||
          normalized.includes('i speak english you reply in bengali')
        ) {
          setLanguageMode('en-bn', true);
          return;
        }

        if (normalized.includes('auto language') || normalized.includes('অটো ভাষা')) {
          setLanguageMode('auto', true);
          return;
        }

        // 3. Action command: "কল হামিদ" / "call hamid"
        if (
          normalized.includes('কল হামিদ') ||
          normalized.includes('হামিদকে কল') ||
          normalized.includes('call hamid')
        ) {
          triggerCallHamid();
          return;
        }

        // 4. Action command: "ওপেন instagram" / "open instagram"
        if (
          normalized.includes('ওপেন instagram') ||
          normalized.includes('ওপেন ইনস্টাগ্রাম') ||
          normalized.includes('open instagram') ||
          (normalized.includes('instagram') && (normalized.includes('open') || normalized.includes('ওপেন')))
        ) {
          triggerOpenInstagram();
          return;
        }

        // 5. Generate answer in matched or bilingual target language
        const replyLang = determineReplyLanguage(trimmed);
        const answerText = generateSmartAnswer(trimmed, replyLang);
        const badge = replyLang === 'en' ? 'AI ANSWER (EN)' : 'AI উত্তর (BN)';
        appendAssistantMessage(answerText, badge, replyLang);
        speakReply(answerText, replyLang);
      }

      }
